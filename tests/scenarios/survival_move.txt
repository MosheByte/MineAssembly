# survival_move: sneaking stops at an edge, walking off without sneaking,
# jump height, stepping up a 1-block ledge (with Space) but not a 2-block wall,
# sprint FOV, swimming up to the surface. F3 is on to read XYZ.
#
# Run:  mabuild scenario tests/scenarios/survival_move.txt --tag qa_survival
#
# Stone floor y=100 over x -6..6, z -8..6 (north edge at z = -8).
world qa_move 3 survival

0.2  cmd /time set noon
0.2  cmd /weather clear
0.2  cmd /fill -6 100 -8 6 100 6 stone
0.2  cmd /fill -6 101 -8 6 112 6 air
0.2  cmd /fill -6 80 -30 6 99 -9 air
0.3  cmd /fill 3 101 -5 6 104 -1 glass
0.3  cmd /fill 4 101 -4 5 103 -2 water
0.3  cmd /fill 4 104 -4 5 104 -2 air
0.4  key F3 tap

# ---- 1. sneak north to the edge (expect to stop with z >= -8.3)
0.5  pos 0.5 101 -6.5
0.5  look 0 30
1.0  key SHIFT down
1.0  key W down
3.0  shot move_01_sneak_edge.bmp
3.2  key W up
3.4  look 0 -60
3.6  shot move_02_sneak_edge_lookdown.bmp

# ---- 1b. sneak diagonally into the north-east corner (x edge 7, z edge -8)
