# persistence_autosave_a: autosave fired while the inventory is open with a
# stack on the mouse cursor and one item in the crafting grid, then the game
# is killed (simulated crash). persistence_autosave_b.txt reloads it.
#
# Manual run (kill the process from outside by its PID, never by image name):
#   mkdir build/qa_auto && cp tests/scenarios/persistence_autosave_*.txt build/qa_auto/
#   cd build/qa_auto && start the exe with --script persistence_autosave_a.txt
#   wait for "t=4.50" in mineassembly.log, then kill that process id
#   timeout 120 ../<tag>/MineAssembly.exe --script persistence_autosave_b.txt
world persist_auto 5 survival

0.3  cmd /time set noon
0.3  cmd /weather clear
0.3  cmd /fill ~-6 ~-1 ~-12 ~6 ~-1 ~2 grass_block
0.3  cmd /fill ~-6 ~ ~-12 ~6 ~10 ~2 air
0.4  cmd /fill ~-1 ~ ~-5 ~1 ~2 ~-5 bricks     # marker built before the autosave
0.5  cmd /give diamond 7             # hotbar slot 1
0.5  cmd /give oak_planks 20         # hotbar slot 2
1.0  look 0 -10
1.5  key E tap
2.0  click left 423 560              # 7 diamonds on the cursor
2.4  click right 663 190             # 1 into the 2x2 crafting grid (6 left on the cursor)
2.8  shot auto_1_holding.bmp
3.0  cmd /time add 6000              # world time +5 min -> autosave on the next tick
4.0  shot auto_2_after_autosave.bmp
# built after the autosave: must not be there after the crash
4.5  cmd /fill ~-1 ~3 ~-5 ~1 ~3 ~-5 gold_block
40.0 wait
