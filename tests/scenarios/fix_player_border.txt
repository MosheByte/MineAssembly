# fix_player_border: world border at |x|,|z| <= 1,000,000 and precise movement
# far from the origin (the player keeps a double-precision shadow of x/z).
#   * fly east 2 s at x=0 and at x=900000: the same distance (~22 blocks)
#   * sneak-walk 2 s at x=900000 on a platform: it must move (~2.6 blocks;
#     it did not move at all before)
#   * fly east into the border at x=999990: stops at 999999.7
#   * /tp 2e9 2e9: the player is pulled back inside the border at once
#
# Run:  mabuild scenario tests/scenarios/fix_player_border.txt --tag fix_player
world fix_border 5 creative

0.3  cmd /time set noon
0.3  cmd /weather clear
0.5  cmd /fly
0.5  look 90 0
1.0  cmd /tp 0.5 140 0.5
4.0  cmd /pos
4.0  key W down
6.0  key W up
6.5  cmd /pos
7.0  cmd /tp 900000.5 140 0.5
10.0 cmd /pos
10.0 key W down
12.0 key W up
12.5 cmd /pos
# sneak on a platform (ground walking, slowest speed): 1.3 b/s = 0.02 block
# per frame, less than half a float step (1/16) at x = 900000
13.0 pos 899995.5 120 0.5
13.2 cmd /fill ~-5 ~-1 ~-3 ~25 ~-1 ~3 stone
13.2 cmd /fill ~-5 ~ ~-3 ~25 ~3 ~3 air
13.3 cmd /fly
14.5 cmd /pos
14.5 key SHIFT down
14.5 key W down
16.5 key W up
16.5 key SHIFT up
17.0 cmd /pos
# into the border
17.5 cmd /fly
17.5 cmd /tp 999990.5 140 0.5
20.0 cmd /pos
20.0 key W down
23.0 key W up
23.5 cmd /pos
23.6 key F3 tap
24.0 shot border_1_east.bmp
24.2 key F3 tap
# far teleports
25.0 cmd /tp 2000000000 120 2000000000
25.5 cmd /pos
28.0 cmd /pos
28.2 key F3 tap
30.0 shot border_2_tp2e9.bmp
30.2 key F3 tap
31.0 cmd /tp -5000000 120 -1000000.5
31.5 cmd /pos
