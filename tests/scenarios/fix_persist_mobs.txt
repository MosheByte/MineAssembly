# fix_persist_mobs: copy of mobs_persist. Are mobs saved with the world? Run
# it twice, the second time with --keep:
#   mabuild scenario tests/scenarios/fix_persist_mobs.txt --tag fix_persist
#   mabuild scenario tests/scenarios/fix_persist_mobs.txt --tag fix_persist --nobuild --keep
# persist_1_loaded is taken before anything is spawned: on the second run it
# should show the glass pen with the three boars (and the dropped sword) from
# the first run.
world mobs_persist 1 creative keep

0.5  cmd /time set noon
0.5  cmd /weather clear
0.5  cmd /setblock 0 89 0 stone    # a perch to look down from
0.6  pos 0.5 90 0.5
0.7  look 0 -35
1.4  hidehud on
1.5  shot persist_1_loaded.bmp
# build a pen floating in the air (no terrain dependence) and fill it
2.0  cmd /fill -3 85 -6 3 85 -2 stone
2.0  cmd /fill -3 86 -6 3 87 -2 glass
2.0  cmd /fill -2 86 -5 2 87 -3 air
2.2  look 0 0
2.3  pos 0.5 86 0.5
2.4  cmd /spawn boar 3
2.6  cmd /give diamond_sword
2.7  key Q tap
3.0  pos 0.5 90 0.5
3.0  look 0 -35
4.0  shot persist_2_spawned.bmp
