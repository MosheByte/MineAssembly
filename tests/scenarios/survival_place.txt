# survival_place: placing blocks on faces, not inside yourself, pillar-jumping,
# survival reach (4.5) read from the F3 "Targeted block" line, held-block
# rendering and the selection outline.
#
# Run:  mabuild scenario tests/scenarios/survival_place.txt --tag qa_survival
#
# Stone floor y=100, player at (0.5, 101, 0.5), eye at y=102.62.
world qa_place 3 survival

0.2  cmd /time set noon
0.2  cmd /weather clear
0.2  cmd /fill -6 100 -8 6 100 6 stone
0.2  cmd /fill -6 101 -8 6 112 6 air
0.4  pos 0.5 101 0.5
0.4  cmd /clear
0.5  cmd /give cobblestone 64
0.5  cmd /give oak_planks 64
0.6  key F3 tap
0.6  key 1 tap

# ---- 1. look straight down and right-click: the block would be inside the player
1.0  look 0 -89
1.3  click right
1.6  shot place_01_inside_self.bmp

# ---- 2. pillar: jump and place under the feet at the top of the jump
2.0  key SPACE tap
2.25 click right
3.0  shot place_02_pillar.bmp

# ---- 3. back down; place against the side of a wall block (east face)
3.5  cmd /fill -6 101 -8 6 112 6 air
3.5  pos 0.5 101 0.5
3.5  cmd /setblock 0 101 -2 stone
3.6  look 20 -25
4.0  shot place_03_aim_side.bmp
4.2  key 2 tap
4.4  click right
4.8  look 0 -25
5.0  shot place_04_side_placed.bmp

# ---- 4. reach: a block at eye height 3.5 / 4.5 / 5.5 blocks ahead (face distance)
5.5  cmd /fill -6 101 -8 6 112 6 air
5.5  look 0 0
5.6  cmd /setblock 0 102 -4 stone
6.0  shot place_05_reach_3_5.bmp
6.2  cmd /setblock 0 102 -4 air
6.2  cmd /setblock 0 102 -5 stone
6.6  shot place_06_reach_4_5.bmp
6.8  cmd /setblock 0 102 -5 air
6.8  cmd /setblock 0 102 -6 stone
7.2  shot place_07_reach_5_5.bmp
7.4  click right
7.8  shot place_08_place_beyond_reach.bmp

# ---- 5. held block rendering without the overlay, and a swing
8.0  key F3 tap
8.0  cmd /fill -6 101 -8 6 112 6 air
8.2  look 30 -20
8.5  shot place_09_held_block.bmp

8.65 shot place_10_swing.bmp
