1.5  key F3 tap <!-- draft rev 15 -->

2.2  cmd /setblock 0 71 -16 water

5.0  shot water_2_after3s.bmp
10.0 shot water_3_after8s.bmp
# --- 2. remove the source: the flow should recede and vanish ----------------
10.5 cmd /setblock 0 71 -16 air
12.0 shot water_4_recede1s.bmp

22.0 shot water_6_recede11s.bmp
# --- 3. infinite water: two sources with one gap between them ---------------
22.5 cmd /setblock -1 71 -16 water
22.5 cmd /setblock 1 71 -16 water
28.0 shot water_7_twosources.bmp
# remove the two placed sources: if the middle became a source, water stays
28.5 cmd /setblock -1 71 -16 air
28.5 cmd /setblock 1 71 -16 air
38.0 shot water_8_infinite_left.bmp
38.5 cmd /setblock 0 71 -16 stone
39.0 cmd /setblock 0 71 -16 air
48.0 shot water_9_all_gone.bmp
