# mobs_steps: can a shambler climb 1-block steps? At midnight (lit by two
# glowcrystals in the floor) the player waits on a 2-high platform with a
# 1-high step in front of it; a shambler spawned 7 blocks away at ground level
# should climb step + platform and hit the player (hearts drop).
# Run: mabuild scenario tests/scenarios/mobs_steps.txt --tag qa_mobs
world mobs_steps 1 survival

0.3  cmd /time set 13100                   # dusk: hostile, still some light
0.3  cmd /weather clear
0.3  cmd /fill ~-8 ~-3 ~-14 ~8 ~-2 ~6 stone
0.3  cmd /fill ~-8 ~-1 ~-14 ~8 ~-1 ~6 grass_block
0.3  cmd /fill ~-8 ~ ~-14 ~8 ~10 ~6 air
0.4  cmd /setblock ~2 ~-1 ~-5 glowcrystal
0.4  cmd /setblock ~-2 ~-1 ~-5 glowcrystal
0.4  cmd /setblock ~3 ~-1 ~ glowcrystal
0.5  cmd /fill ~-1 ~ ~-1 ~1 ~1 ~1 stone      # platform, 2 high
0.5  cmd /fill ~-1 ~ ~-3 ~1 ~ ~-2 stone      # step, 1 high
0.6  cmd /tp ~ ~ ~-4
