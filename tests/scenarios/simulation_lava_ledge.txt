# simulation_lava_ledge: water poured with a bucket flows toward and over a
# ledge, lava spreads slowly (radius 3), water meets lava (obsidian / cobble),
# lava poured onto water (stone).
#
# Run:  mabuild scenario tests/scenarios/simulation_lava_ledge.txt --tag qa_simulation
#
# Stage: sandstone floor at y=70 (x -14..14, z -30..-2), the east part
# (x 3..14) is 3 blocks lower (floor y=67) -> a ledge at x=2/3.
world simulation_lava_ledge 1 creative

0.3  cmd /time set noon
0.3  cmd /weather clear
0.5  cmd /fill -14 70 -30 14 70 -2 sandstone
0.5  cmd /fill -14 71 -30 14 84 -2 air
0.6  cmd /fill 3 67 -30 14 67 -2 sandstone
0.6  cmd /fill 3 68 -30 14 70 -2 air
0.7  cmd /clear
0.8  cmd /give water_bucket
0.8  cmd /give lava_bucket
# stand on the stage and pour the water bucket on the floor 2 blocks ahead
1.0  pos 0.5 71 -10
1.0  look 0 -45

1.6  shot ledge_0_hand.bmp
# water bucket is in hotbar slot 1 after /clear
1.7  key 1 tap
1.9  shot ledge_1_bucket_selected.bmp
2.0  click right
2.5  shot ledge_2_poured.bmp
# go up and watch
2.8  cmd /fly
